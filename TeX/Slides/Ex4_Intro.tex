\begin{frame}<0>[label=Ex4,fragile]{Some guidance about the session}
    \vspace{-2mm}
    \begin{itemize}
        \item You have roughly 60 minutes\\
              \begin{onlyenv}<1>
                  \then You won't manage to do everything! \Remark{That's not expected either!}
              \end{onlyenv}
              \begin{onlyenv}<2>
                  \then You should manage to do the new exercises\\
                  \then If you are done earlier, do the rest!
              \end{onlyenv}
        \item Each exercise should be solved in a dedicated script
              \PB{\Remark{Use the terminal to experiment, if needed}}
        \item<only@1> \PQ{For the moment you can only tackle exercises 1, 2 and 5}
        \item<only@1> Start with exercise 2 but after 30 minutes switch to either exercise 1 or 5\\
                      \then Exercise 1 is more about \bash|wait| and possibly HPC oriented\\
                      \then Exercise 5 is more about understanding concurrency\\
                      \then \PS{It would be instructive to try to do both} \Remark{if not now in the next session}
        \item<only@2> \PQ{Now you can tackle all exercises}
        \item<only@2> You can also finish up the work from the last practice session\\
                      \then Consider this as further practice on this topic
        \item Note down your questions, doubts, curiosities, \ldots
        \item \alert{\textbf{If you can't resist:} Give your solution to an AI to get feedback} \Remark{The trainer can also give you feedback}
    \end{itemize}
    \begin{varblock}{}[0.62\textwidth]{Come up with suggestions!}
        In the closing sessions we'll solve an exercise together.
        If you have any preference, feel free to speak up.
    \end{varblock}
\end{frame}
