\begin{frame}<0>[label=ControlFlow,fragile]
    {Displaying a grid}{Write a script that displays a grid with a ticked cell}
    \vspace{-1mm}
    \begin{enumerate}
        \item Let the user choose a grid size among \bash|3x3|, \bash|5x5| and \bash|7x7|
        \item Pick a row and a column randomly
        \item Impose that cells on the diagonal cannot be picked
        \item Print the grid mark the chosen position with an \bash|X|.
    \end{enumerate}
    \begin{varblock}{example}[0.85\textwidth]{How could it look like?}
        For example, choosing a \bash|5x5| grid and coordinates  \bash|(3,2)| could produce:
        \begin{lstlisting}[style=MyBash, numbers=none, aboveskip=3mm, belowskip=-5mm, xrightmargin=10mm, xleftmargin=10mm]
            $ ./display-grid.bash

            # Interaction with the user to pick 5x5

            |+  1 2 3 4 5
            1 . . . . .
            2 . . . . .
            3 . X . . .
            4 . . . . .
            5 . . . . .+|
        \end{lstlisting}
    \end{varblock}
    \FrameRemark{\alert{You will expand this script later in the course. Be kind to your future self!}}
\end{frame}

\begin{frame}<0>[label=StoreGrid,fragile]{Using our first arrays}{Store a 2D grid into an array}
    \vspace{-2mm}
    \alert{The grid was actually a map!}
    \textbf{Ancient explorers marked valuable locations at prime-numbered cells.}
    \begin{enumerate}
        \item Start back from the result of your last practice session or from this script
        \item Store the grid into an array \bash|grid| of size $N_x\,N_y$ with only dots \bash|'.'|
        \item Store in an other array \bash|gold| the prime numbers up to $N_x\cdot N_y$ (hardcoding them)\\
              \then For the \texttt{3x3} this is \bash|gold=(2 3 5 7)|
        \item Change dots in \bash|grid| with \bash|'X'| at indices stored in \bash|gold|
        \item Draw the two coordinates $(i,j)$ of a random \textbf{\PP{free}} grid point and change its dot to \bash|'G'|
              \then The 2D lattice is stored with a super-index $s$ unrolled:\\
              \phantom{\then} $s = (i-1) + (j-1)\cdot N_x$ with $(i,j) \in [1,N_x]\times[1,N_y]$
        \item Print the resulting \bash|grid|, e.g.
              \begin{lstlisting}[style=MyBash, numbers=none, aboveskip=3mm, belowskip=-5mm, xrightmargin=15mm]
                  |+  1 2 3+|
                  |+1 . . X+|   # Make sure that the G cell does not
                  |+2 X . X+|   # replace any cell already marked by X
                  |+3 . X G+|
              \end{lstlisting}
    \end{enumerate}
\end{frame}
